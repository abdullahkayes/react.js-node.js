<!-- JavaScript Logic for Minimize & Restore -->
<script>
  function minimizeModal() {
    const backdrop = document.getElementById('verification-modal-backdrop');
    const minimizedBar = document.getElementById('verification-minimized-bar');
    
    // Hide full backdrop/modal
    backdrop.classList.add('hidden');
    
    // Show floating bar at bottom right
    minimizedBar.classList.remove('hidden');
    minimizedBar.classList.add('flex');
  }

  function restoreModal() {
    const backdrop = document.getElementById('verification-modal-backdrop');
    const minimizedBar = document.getElementById('verification-minimized-bar');
    
    // Hide floating bar
    minimizedBar.classList.add('hidden');
    minimizedBar.classList.remove('flex');
    
    // Show full backdrop/modal
    backdrop.classList.remove('hidden');
  }
</script>
